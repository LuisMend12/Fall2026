\textbf{Steps.}
\begin{mathpar}
\inferrule*[right=StepIfSearch]
  {e_1 \step e_1'}
  {\ite{e_1}{e_2}{e_3} \step \ite{e_1'}{e_2}{e_3}}
\\
\inferrule*[right=StepIfTrue]
  { }
  {\ite{\tru}{e_2}{e_3} \step e_2}
\and
\inferrule*[right=StepIfFalse]
  { }
  {\ite{\fls}{e_2}{e_3} \step e_3}
\end{mathpar}
Only the condition is searched; the branches are never evaluated until one is selected.
\end{task}

\begin{task}{Task 2}
\begin{align*}
&\ite{1+2=4}{\str{Fake news}}{\str{True fact}} \\
\step\ &\ite{3=4}{\str{Fake news}}{\str{True fact}}
  && \text{(\rn{StepIfSearch})}\\
\step\ &\ite{\fls}{\str{Fake news}}{\str{True fact}}
  && \text{(\rn{StepIfSearch})}\\
\step\ &\str{True fact}
  && \text{(\rn{StepIfFalse})}
\end{align*}
The derivations of the first two steps are:
{\scriptsize
\begin{mathpar}
\inferrule*[right=StepIfSearch]
  {\inferrule*[right=StepSearchEqLeft]
    {\inferrule*[right=StepAdd]{ }{1+2 \step 3}}
    {1+2=4 \step 3=4}}
  {\ite{1+2=4}{\str{Fake news}}{\str{True fact}} \step \ite{3=4}{\str{Fake news}}{\str{True fact}}}
\\
\inferrule*[right=StepIfSearch]
  {\inferrule*[right=StepNotEq]{3 \neq 4}{3=4 \step \fls}}
  {\ite{3=4}{\str{Fake news}}{\str{True fact}} \step \ite{\fls}{\str{Fake news}}{\str{True fact}}}
\end{mathpar}
}
and $\str{True fact} \val$, so evaluation is complete.
\end{task}

\begin{task}{Task 4}
\textbf{Preservation.} If $e:\tau$ and $e \step e'$, then $e':\tau$.

\emph{Proof.} By induction on the derivation of $e \step e'$. New cases:

\begin{itemize}[leftmargin=*]
\item \textbf{Case \rn{StepIfTrue}.} Then $e = \ite{\tru}{e_2}{e_3}$ and $e' = e_2$.
  The only typing rule whose conclusion has the form $\ite{\cdot}{\cdot}{\cdot}$ is
  \rn{TypeIf}, so by inversion $\tru:\tbool$, $e_2:\tau$, and $e_3:\tau$.
  In particular $e' = e_2 : \tau$.

\item \textbf{Case \rn{StepIfFalse}.} Then $e = \ite{\fls}{e_2}{e_3}$ and $e' = e_3$.
  By inversion on \rn{TypeIf}, $e_3 : \tau$, i.e.\ $e':\tau$.

\item \textbf{Case \rn{StepIfSearch}.} Then $e = \ite{e_1}{e_2}{e_3}$,
  $e' = \ite{e_1'}{e_2}{e_3}$, and $e_1 \step e_1'$.
  By inversion on \rn{TypeIf}, $e_1:\tbool$, $e_2:\tau$, and $e_3:\tau$.
  Since $e_1 : \tbool$ and $e_1 \step e_1'$, the induction hypothesis gives $e_1' : \tbool$.
  Then by \rn{TypeIf}, $\ite{e_1'}{e_2}{e_3} : \tau$, i.e.\ $e':\tau$. \qed
\end{itemize}
\end{task}

\begin{task}{Task 6}
\textbf{Progress.} If $e:\tau$, then $e \val$ or there exists $e'$ such that $e \step e'$.

\emph{Proof.} By induction on the derivation of $e:\tau$. New cases:

\begin{itemize}[leftmargin=*]
\item \textbf{Case \rn{TypeTrue}.} Then $e = \tru$, and $\tru \val$ by \rn{ValTrue}.

\item \textbf{Case \rn{TypeFalse}.} Then $e = \fls$, and $\fls \val$ by \rn{ValFalse}.

\item \textbf{Case \rn{TypeIf}.} Then $e = \ite{e_1}{e_2}{e_3}$ with
  $e_1:\tbool$, $e_2:\tau$, $e_3:\tau$. By the induction hypothesis on $e_1:\tbool$,
  either $e_1 \val$ or $e_1 \step e_1'$ for some $e_1'$.
  \begin{itemize}
  \item If $e_1 \step e_1'$, then by \rn{StepIfSearch},
    $e \step \ite{e_1'}{e_2}{e_3}$.
  \item If $e_1 \val$, then since $e_1 : \tbool$, Canonical Forms (Task 5) gives
    $e_1 = \tru$ or $e_1 = \fls$.
    If $e_1 = \tru$, then $e \step e_2$ by \rn{StepIfTrue}.
    If $e_1 = \fls$, then $e \step e_3$ by \rn{StepIfFalse}.
  \end{itemize}
  In every subcase $e$ steps. \qed
\end{itemize}
\end{task}

\section*{Division by zero}

\begin{task}{Task 7}
\begin{mathpar}
\inferrule*[right=TypeNum]{ }{n : \mz}
\and
\inferrule*[right=TypeNumNZ]{n \neq 0}{n : \nz}
\\
\inferrule*[right=TypeAdd]{e_1:\mz \\ e_2:\mz}{e_1 + e_2 : \mz}
\and
\inferrule*[right=TypeSub]{e_1:\mz \\ e_2:\mz}{e_1 - e_2 : \mz}
\\
\inferrule*[right=TypeMulNZ]{e_1:\nz \\ e_2:\nz}{e_1 * e_2 : \nz}
\and
\inferrule*[right=TypeMulMZ]{e_1:\mz \\ e_2:\mz}{e_1 * e_2 : \mz}
\\
\inferrule*[right=TypeDiv]{e_1:\mz \\ e_2:\nz}{e_1 / e_2 : \mz}
\end{mathpar}
Justification: the sums and differences of nonzero integers can be zero (e.g.\ $1 + (-1)$, $1-1$),
so the addition and subtraction can only give $\mz$. A product of two nonzero integers is nonzero,
but a product with a possibly-zero factor may be zero. For division, the divisor must be
$\nz$ to avoid dividing by zero; the result is $\mz$ because integer division can produce
zero even with a nonzero dividend (e.g.\ $1/2 = 0$).
\end{task}

%--------------------------------------------------------------------
\begin{task}{Task 8}
\textbf{Progress.} If $e:\tau$, then $e \val$ or there exists $e'$ such that $e \step e'$.

\emph{Proof.} By induction on the derivation of $e:\tau$.
\begin{itemize}[leftmargin=*]
\item \textbf{Case \rn{TypeNum}.} Then $e = n$, and $n \val$ by \rn{ValNum}.

\item \textbf{Case \rn{TypeNumNZ}.} Then $e = n$ (with $n \neq 0$), and $n \val$ by \rn{ValNum}.

\item \textbf{Case \rn{TypeDiv}.} Then $e = e_1 / e_2$ with $e_1:\mz$ and $e_2:\nz$.
  By the induction hypothesis on $e_1 : \mz$, either $e_1 \step e_1'$ or $e_1 \val$.
  \begin{itemize}
  \item If $e_1 \step e_1'$, then $e_1/e_2 \step e_1'/e_2$ by \rn{StepSearchOpLeft}
    (with $op = /$).
  \item If $e_1 \val$, then by Canonical Forms (case 2), $e_1 = n_1$ for some $n_1$.
    By the induction hypothesis on $e_2:\nz$, either $e_2 \step e_2'$ or $e_2 \val$.
    \begin{itemize}
    \item If $e_2 \step e_2'$, then $n_1/e_2 \step n_1/e_2'$ by \rn{StepSearchOpRight}
      (with $op = /$).
    \item If $e_2 \val$, then since $e_2 : \nz$, Canonical Forms (case 1) gives
      $e_2 = n_2$ for some $n_2 \neq 0$. Hence the premise of \rn{StepDiv} holds, and
      $n_1 / n_2 \step n$ where $n$ is the integer quotient of $n_1$ by $n_2$.
    \end{itemize}
  \end{itemize}
  In every subcase $e$ steps. \qed
\end{itemize}

\bigskip
\textbf{Preservation.} If $e:\tau$ and $e \step e'$, then $e':\tau$.

\emph{Proof.} By induction on the derivation of $e \step e'$.
\begin{itemize}[leftmargin=*]
\item \textbf{Case \rn{StepAdd}.} Then $e = n_1 + n_2$ and $e' = n$ where $n = n_1 + n_2$.
  The only typing rule whose conclusion has the form $e_1 + e_2$ is \rn{TypeAdd},
  so by inversion $\tau = \mz$. By \rn{TypeNum}, $n : \mz$, i.e.\ $e' : \tau$.

\item \textbf{Case \rn{StepSub}.} Then $e = n_1 - n_2$ and $e' = n$ where $n = n_1 - n_2$.
  The only typing rule for $e_1 - e_2$ is \rn{TypeSub}, so by inversion $\tau = \mz$.
  By \rn{TypeNum}, $n : \mz$.

\item \textbf{Case \rn{StepDiv}.} Then $e = n_1 / n_2$ with $n_2 \neq 0$ and $e' = n$, the
  integer quotient of $n_1$ by $n_2$. The only typing rule for $e_1 / e_2$ is
  \rn{TypeDiv}, so by inversion $\tau = \mz$. By \rn{TypeNum}, $n : \mz$. \qed
\end{itemize}
\end{task}

%--------------------------------------------------------------------
\begin{task}{Bonus Task 1}
\textbf{Case \rn{StepMul}.} Then $e = n_1 * n_2$ and $e' = n$ where $n = n_1 n_2$.
Two typing rules have a conclusion of the form $e_1 * e_2$, so by inversion there are two cases.
\begin{itemize}[leftmargin=*]
\item \textbf{Subcase \rn{TypeMulMZ}.} Then $\tau = \mz$, and $n : \mz$ by \rn{TypeNum}.
\item \textbf{Subcase \rn{TypeMulNZ}.} Then $\tau = \nz$, with $n_1 : \nz$ and $n_2 : \nz$.
  Since $n_1 \val$ and $n_2 \val$ (by \rn{ValNum}), Canonical Forms (case 1) gives
  $n_1 \neq 0$ and $n_2 \neq 0$. The product of two nonzero integers is nonzero, so
  $n = n_1 n_2 \neq 0$, and by \rn{TypeNumNZ}, $n : \nz$. \qed
\end{itemize}
\end{task}

\begin{task}{Task 9}
\textbf{Expression.} $1 / (1 + 1)$.

This never divides by zero (it evaluates to $1/2 \step 0$), but it is not well-typed: the
only rule for $/$ is \rn{TypeDiv}, which needs $1 + 1 : \nz$, and the only rule for $+$
(\rn{TypeAdsd}) gives type $\mz$.

\textbf{Suggested refinement.} Track signs. Replace $\nz$ with two more precise types
$\kw{pos}$ (strictly positive) and $\kw{neg}$ (strictly negative), both of which count as
nonzero (e.g.\ $\kw{pos} \le \nz$ and $\kw{neg} \le \nz$ in a subtyping sense). Then add
rules such as
\begin{mathpar}
\inferrule{n > 0}{n : \kw{pos}}
\and
\inferrule{e_1:\kw{pos} \\ e_2:\kw{pos}}{e_1 + e_2 : \kw{pos}}
\and
\inferrule{e_1:\kw{neg} \\ e_2:\kw{neg}}{e_1 + e_2 : \kw{neg}}
\and
\inferrule{e_1:\kw{pos} \\ e_2:\kw{neg}}{e_1 - e_2 : \kw{pos}}
\end{mathpar}
together with the usual sign rules for multiplication
($\kw{pos}*\kw{neg} : \kw{neg}$, etc.). Now $1 + 1 : \kw{pos}$, hence $1 + 1 : \nz$, and
$1/(1+1) : \mz$ by \rn{TypeDiv}. Division by zero is still ruled out, since every
$\kw{pos}$ or $\kw{neg}$ expression evaluates to a nonzero integer. (This still rejects
some safe programs, such as $1/(2-1)$; a more precise system could track exact values or
intervals, e.g.\ refinement types like $\{x : \tint \mid x \geq 1\}$.)
\end{task}
