\section*{Problem 1 (10 points)}
Prove that for any $x,y\in\mathbb{R}$,
\[
  \bigl|\,|x|-|y|\,\bigr| \le |x-y| .
\]

\begin{proof}
We start from the identity
\[
  x = (x-y) + y .
\]
Taking absolute values and applying the triangle inequality,
\[
  |x| = |(x-y)+y| \le |x-y| + |y| .
\]
Subtracting $|y|$ from both sides gives
\[
  |x| - |y| \le |x-y| .
\]
We can do the same thing starting from the identity
\[
  y = (y-x) + x .
\]
Taking absolute values and applying the triangle inequality,
\[
  |y| = |(y-x)+x| \le |y-x| + |x| .
\]
Subtracting $|x|$ from both sides gives
\[
  |y| - |x| \le |y-x| .
\]
Since $y-x = -(x-y)$ and $|-t| = |t|$ for every real $t$, we have
$|y-x| = |x-y|$.

Now let $a = |x|-|y|$ and $c = |x-y|$. Note that $c \ge 0$.
The first inequality says $a \le c$. In the second, the left-hand side is
$|y|-|x| = -a$ and the right-hand side is $|y-x| = c$, so it says
$-a \le c$, i.e.\ $a \ge -c$. Together,
\[
  -c \le a \le c .
\]
Since $c \ge 0$, this is equivalent to $|a| \le c$, that is,
\[
  \bigl|\,|x|-|y|\,\bigr| \le |x-y| ,
\]
which is the original statement.

\end{proof}

\section*{Problem 2 (10 points)}
Use the limit definition to compute
\[
  \lim_{n\to\infty}\frac{n+2}{3n+5}.
\]

\begin{claim}
$\displaystyle\lim_{n\to\infty}\frac{n+2}{3n+5} = \dfrac{1}{3}$.
\end{claim}

\begin{proof}
First we simplify the distance to $\tfrac13$. Using the common
denominator $3(3n+5) = 9n+15$,
\[
  \frac{n+2}{3n+5} - \frac13
  = \frac{3(n+2) - (3n+5)}{3(3n+5)}
  = \frac{3n+6-3n-5}{9n+15}
  = \frac{1}{9n+15}.
\]
Since this is positive for every $n\in\mathbb{N}$, and $9n+15 > 9n$,
\[
  \left|\frac{n+2}{3n+5} - \frac13\right| = \frac{1}{9n+15} < \frac{1}{9n}.
\]
Now let $\varepsilon > 0$. By the Archimedean property, choose
$N \in \mathbb{N}$ with $N > \dfrac{1}{9\varepsilon}$, so that
$\dfrac{1}{9N} < \varepsilon$. Then for all $n \ge N$,
\[
  \left|\frac{n+2}{3n+5} - \frac13\right|
  = \frac{1}{9n+15}
  < \frac{1}{9n}
  \le \frac{1}{9N}
  < \varepsilon .
\]
Hence $\displaystyle\lim_{n\to\infty}\frac{n+2}{3n+5} = \frac13$.

\end{proof}

\section*{Problem 3 (10 points)}
Prove that if $\displaystyle\lim_{n\to\infty}x_n=\infty$ and $(y_n)_n$ is
bounded, then
\[
  \lim_{n\to\infty}(x_n+y_n)=\infty .
\]

\begin{proof}
Since $(y_n)$ is bounded, there is $M>0$ with $|y_n|\le M$ for every
$n$, i.e.\ $-M\le y_n\le M$. In particular,
\[
  y_n \ge -M \quad\text{for every } n .
\]
Recall that $x_n\to\infty$ means: for every $H\in\mathbb{R}$ there is
$N\in\mathbb{N}$ such that $x_n > H$ for all $n\ge N$.
Adding $x_n$ to both sides of $y_n\ge -M$ gives, for every $n$,
\[
  x_n + y_n \ge x_n - M .
\]
So to make $x_n+y_n > K$, it is enough to make $x_n - M > K$.

Let $K\in\mathbb{R}$ be arbitrary.
We want $x_n - M > K$; solving for $x_n$, this means $x_n > K + M$.
Since $x_n\to\infty$, applying the definition with the height $H = K+M$
gives $N\in\mathbb{N}$ such that
\[
  x_n > K + M \quad\text{for all } n\ge N .
\]
Then for all $n\ge N$,
\[
  x_n + y_n \;\ge\; x_n - M \;>\; (K+M) - M \;=\; K .
\]
Since $K$ was arbitrary, $\displaystyle\lim_{n\to\infty}(x_n+y_n)=\infty$.

\end{proof}
