\section*{Problem 1 (10 points)}

Prove
\[
  p(x,y\mid z) = p(x\mid z)\,p(y\mid x,z)
\]
and also
\[
  p(x\mid y,z) = \frac{p(y\mid x,z)\,p(x\mid z)}{p(y\mid z)}.
\]

\textbf{First identity.} We can apply the ordinary product rule to get $p(u,v) = p(u\mid v)\,p(v)$,
inside the world where $z$ is already given -- i.e.\ treat $p(\cdot \mid z)$ as an
ordinary distribution and apply the product rule to $x$ and $y$ within it, conditioning
on $x$ first:
\[
  p(x,y \mid z) = p(x \mid z)\, p(y \mid x, z).
\]

\textbf{Second identity.} Apply the product rule again to the same quantity
$p(x,y\mid z)$, but conditioning on $y$ first instead:
\[
  p(x,y \mid z) = p(y \mid z)\, p(x \mid y, z).
\]
The left-hand sides of the two expansions are identical, so the right-hand sides
must be equal to each other:
\[
  p(x \mid z)\, p(y \mid x, z) = p(y \mid z)\, p(x \mid y, z).
\]
Dividing both sides by $p(y\mid z)$ (assuming $p(y\mid z) \neq 0$) and solving for
$p(x\mid y,z)$ gives
\[
  p(x \mid y, z) = \frac{p(y \mid x, z)\, p(x \mid z)}{p(y \mid z)},
\]
which is exactly the second identity. $\blacksquare$

\item Factor $P(A)$: $k = 2$ (A is binary), parents = none so $r = 1$ $\to$ cost $= 1*(2-1) = 1$.
\item Factor $P(B\mid A)$: $k = 2$ (B is binary), parents $=\{A\}$ so $r = 2$ (A has 2 values) $\to$ cost $= 2*(2-1) = 2$.

Total parameters $= 1 + 2 + 4 + 4 = 11$.

\subsection*{(b) $p(d)$, $p(d\mid s=\mathrm{tr})$, $p(d\mid s=\mathrm{fa})$}

For this question I used the following factorization of the joint distribution, which is consistent with the network structure:
\[
  p(a,t,s,l,b,e,x,d) = p(a)\,p(t\mid a)\,p(s)\,p(l\mid s)\,p(b\mid s)\,p(e\mid t,l)\,p(x\mid e)\,p(d\mid e,b).
\]


\begin{algorithm}
\caption{Marginal and conditional queries on the Chest Clinic network}
\begin{algorithmic}[1]
\Procedure{JointProb}{$a,t,s,l,b,e,x,d$}
  \State \Return $p(a)\cdot p(t\mid a)\cdot p(s)\cdot p(l\mid s)\cdot p(b\mid s)\cdot p(e\mid t,l)\cdot p(x\mid e)\cdot p(d\mid e,b)$
\EndProcedure
\Procedure{MarginalD}{}
  \State $p_d[\mathrm{tr}] \gets 0$, \quad $p_d[\mathrm{fa}] \gets 0$
  \For{each assignment of $a,t,s,l,b,e,x,d$}
    \State $p_d[d] \gets p_d[d] + \Call{JointProb}{a,t,s,l,b,e,x,d}$
  \EndFor
  \State \Return $p_d$
\EndProcedure
\Procedure{ConditionalDGivenS}{$s^\ast$}
  \State $u[\mathrm{tr}] \gets 0$, \quad $u[\mathrm{fa}] \gets 0$
  \For{each assignment of $a,t,s,l,b,e,x,d$ with $s = s^\ast$}
    \State $u[d] \gets u[d] + \Call{JointProb}{a,t,s,l,b,e,x,d}$
  \EndFor
  \State \Return $u$ normalized so $u[\mathrm{tr}] + u[\mathrm{fa}] = 1$
\EndProcedure
\end{algorithmic}
\end{algorithm}
